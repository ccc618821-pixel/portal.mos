PRAGMA foreign_keys = ON;
DROP TABLE IF EXISTS order_items;
DROP TABLE IF EXISTS orders;
DROP TABLE IF EXISTS clients;
DROP TABLE IF EXISTS configurations;
DROP TABLE IF EXISTS models;
-- Модели ноутбуков
CREATE TABLE models (
    model_id    INTEGER PRIMARY KEY AUTOINCREMENT,
    brand       TEXT NOT NULL,
    name        TEXT NOT NULL,
    image_path  TEXT NOT NULL,
    UNIQUE (brand, name)
);
-- Конфигурации
CREATE TABLE configurations (
    config_id   INTEGER PRIMARY KEY AUTOINCREMENT,
    model_id    INTEGER NOT NULL,
    ram_gb      INTEGER NOT NULL CHECK (ram_gb > 0),
    ssd_gb      INTEGER NOT NULL CHECK (ssd_gb > 0),
    price       REAL    NOT NULL CHECK (price >= 0),
    stock_qty   INTEGER NOT NULL DEFAULT 0 CHECK (stock_qty >= 0),
    FOREIGN KEY (model_id) REFERENCES models(model_id),
    UNIQUE (model_id, ram_gb, ssd_gb)
);
-- Клиенты
CREATE TABLE clients (
    client_id   INTEGER PRIMARY KEY AUTOINCREMENT,
    last_name   TEXT NOT NULL,
    first_name  TEXT NOT NULL,
    patronymic  TEXT,
    phone       TEXT NOT NULL,
    email       TEXT
);
-- Заказы
CREATE TABLE orders (
    order_id    INTEGER PRIMARY KEY AUTOINCREMENT,
    client_id   INTEGER NOT NULL,
    order_date  TEXT NOT NULL DEFAULT (datetime('now')),
    status      TEXT NOT NULL DEFAULT 'новый'
                CHECK (status IN ('новый','оплачен','собран','отгружен','отменён')),
    FOREIGN KEY (client_id) REFERENCES clients(client_id)
);
-- Состав заказа
CREATE TABLE order_items (
    order_id    INTEGER NOT NULL,
    config_id   INTEGER NOT NULL,
    quantity    INTEGER NOT NULL CHECK (quantity > 0),
    unit_price  REAL    NOT NULL CHECK (unit_price >= 0),
    PRIMARY KEY (order_id, config_id),
    FOREIGN KEY (order_id)  REFERENCES orders(order_id),
    FOREIGN KEY (config_id) REFERENCES configurations(config_id)
);

SELECT c.config_id, m.brand || ' ' || m.name AS model,
       c.ram_gb, c.ssd_gb, c.price, c.stock_qty
FROM configurations c
JOIN models m ON m.model_id = c.model_id
WHERE c.stock_qty > 0
ORDER BY m.brand, c.price;
SELECT o.order_id, o.order_date,
       c.last_name || ' ' || c.first_name || ' ' || IFNULL(c.patronymic, '') AS client_fio,
       o.status,
       SUM(oi.quantity * oi.unit_price) AS total
FROM orders o
JOIN clients c     ON c.client_id = o.client_id
JOIN order_items oi ON oi.order_id = o.order_id
GROUP BY o.order_id
ORDER BY o.order_date DESC;
